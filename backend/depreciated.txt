			const filePath = getFilePath('webConfig');
			await writeToJSON(filePath, webConfig);

		}

		const filePath = getFilePath('config')
		await writeToJSON(filePath, config);

		return `Updated ${filePath}`
	} catch (err) {
		throw err;
	}
}
